# Point-in-time XLK constituent membership is a leaf dependency with no ONNX,
# Boost or serialization requirements. It lives in its own target so the FNSPID
# importer in services/ml_model can reuse the single implementation without
# pulling in FinBERT/onnxruntime.
add_library(arrakis_xlk_membership src/xlk_membership.cpp)
target_compile_features(arrakis_xlk_membership PUBLIC cxx_std_20)
target_include_directories(arrakis_xlk_membership PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(arrakis_xlk_membership PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()

add_library(arrakis_news_nlp_core src/finbert.cpp src/aggregation.cpp)
target_compile_features(arrakis_news_nlp_core PUBLIC cxx_std_20)
target_include_directories(arrakis_news_nlp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${PROJECT_SOURCE_DIR}/libs/serialization/include)
target_link_libraries(arrakis_news_nlp_core PUBLIC Boost::json arrakis_xlk_membership OpenSSL::Crypto)
find_package(onnxruntime CONFIG QUIET)
if(onnxruntime_FOUND)
    target_compile_definitions(arrakis_news_nlp_core PRIVATE ARRAKIS_HAS_ONNXRUNTIME=1)
    target_link_libraries(arrakis_news_nlp_core PRIVATE onnxruntime::onnxruntime)
else()
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATHS /opt/homebrew/opt/onnxruntime/include /usr/local/include)
    find_library(ONNXRUNTIME_LIBRARY onnxruntime PATHS /opt/homebrew/opt/onnxruntime/lib /usr/local/lib)
    if(ONNXRUNTIME_INCLUDE_DIR AND ONNXRUNTIME_LIBRARY)
        target_compile_definitions(arrakis_news_nlp_core PRIVATE ARRAKIS_HAS_ONNXRUNTIME=1)
        target_include_directories(arrakis_news_nlp_core PRIVATE ${ONNXRUNTIME_INCLUDE_DIR})
        target_link_libraries(arrakis_news_nlp_core PRIVATE ${ONNXRUNTIME_LIBRARY})
    endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(arrakis_news_nlp_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
